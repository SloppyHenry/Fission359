  qml/Main.qml qml/AboutWindow.qml qml/AppMenu.qml qml/BedLabels.qml qml/Btn.qml qml/DocTabs.qml
  qml/Float.qml qml/Focusring.qml qml/HistoryPanel.qml qml/Icon.qml qml/IconBtn.qml qml/InfoRow.qml qml/LeftPanel.qml
  qml/LibraryCard.qml qml/Logo.qml qml/MenuStrip.qml qml/NativeMenuBar.qml qml/NavButton.qml

  qml/ViewOverlay.qml qml/ShapeCatalog.qml qml/ShapeFlyout.qml qml/Toggle.qml qml/HandleLayer.qml qml/GizmoLayer.qml qml/ToolCatalog.qml qml/StyleButton.qml qml/WordField.qml qml/FontField.qml qml/icons.js qml/clearplace.js)

qt_add_qml_module(fission3d
  URI Fission3D
  VERSION 1.0
  QML_FILES ${FISSION_QML_FILES}
  # The module directory "Fission3D" would collide with the executable "fission3d" on
  # case-insensitive file systems (macOS), so it gets its own parent directory.
  OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml-modules/Fission3D")

# The interface font (IBM Plex, SIL Open Font License 1.1) is part of the program.
qt_add_resources(fission3d "fonts"
  PREFIX "/fission/fonts"
  BASE resources/fonts
  FILES
    resources/fonts/IBMPlexSans-Regular.ttf resources/fonts/IBMPlexSans-Medium.ttf
    resources/fonts/IBMPlexSans-SemiBold.ttf resources/fonts/IBMPlexSans-Bold.ttf
    resources/fonts/IBMPlexMono-Regular.ttf resources/fonts/IBMPlexMono-Medium.ttf
    resources/fonts/IBMPlexMono-SemiBold.ttf
    # the styles lettering can use: bold and italic of both families
    resources/fonts/IBMPlexSans-Italic.ttf resources/fonts/IBMPlexSans-BoldItalic.ttf
    resources/fonts/IBMPlexMono-Bold.ttf resources/fonts/IBMPlexMono-Italic.ttf
    resources/fonts/IBMPlexMono-BoldItalic.ttf)

# Texts are English in the source (qsTr); the German translation is a Qt Linguist file.
# `cmake --build <dir> --target update_translations` refreshes the .ts files from the sources.
set(_fission_tr_sources src/main.cpp)
foreach(_qml IN LISTS FISSION_QML_FILES)
  list(APPEND _fission_tr_sources "${CMAKE_CURRENT_SOURCE_DIR}/${_qml}")
endforeach()
qt_add_translations(fission3d
  TS_FILES i18n/fission3d_de.ts i18n/fission3d_en.ts
  SOURCES ${_fission_tr_sources}
  RESOURCE_PREFIX "/i18n"
  LUPDATE_OPTIONS -no-obsolete -locations relative)

# The app wires the layers together. Kernel implementations are linked here only,
# so that no other layer can depend on them.
target_link_libraries(fission3d PRIVATE
  Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml
  fission_core fission_ui fission_services spdlog::spdlog
  fission_geometry_occt fission_geometry_mesh fission_geometry_sketch
  fission_features fission_io fission_analysis)

fission_apply_compiler_options(fission3d)

# The application ID is the name under which the desktop knows the app (Wayland app-id,
# D-Bus, .desktop file); main.cpp reads it from here.
target_compile_definitions(fission3d PRIVATE FISSION_APP_ID="${FISSION_APP_ID}")

# --- Packaging (AP-0.7, ADR-0006) -------------------------------------------------------
# macOS: Fission3D.app is built in the build tree; scripts/package finishes it
# (macdeployqt, ad-hoc signature, .dmg). Linux: `cmake --install` creates a FHS tree that
# scripts/package turns into an AppImage and a Flatpak.
set(FISSION_BUNDLE_EXECUTABLE fission3d)

# vcpkg's Qt has no qmake, but linuxdeploy's Qt plugin asks `qmake -query` where Qt lives. This
# shim answers from the paths CMake found (also correct for apt and Homebrew Qt).
set(_qmlscanner_dir "")
if(TARGET Qt6::qmlimportscanner)
  set(_qmlscanner_dir "$<TARGET_FILE_DIR:Qt6::qmlimportscanner>")
endif()
set(_qt_prefix "${QT6_INSTALL_PREFIX}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/qmake-shim" CONTENT
"#!/bin/sh
# Generated by CMake (apps/fission3d/CMakeLists.txt): answers qmake -query for the packaging tools.
vars() {
cat <<'EOT'
QT_VERSION:${Qt6_VERSION}
QT_INSTALL_PREFIX:${_qt_prefix}
QT_HOST_PREFIX:${_qt_prefix}
QT_INSTALL_ARCHDATA:${_qt_prefix}/${QT6_INSTALL_ARCHDATA}
QT_INSTALL_DATA:${_qt_prefix}/${QT6_INSTALL_DATA}
QT_INSTALL_BINS:${_qt_prefix}/${QT6_INSTALL_BINS}
QT_HOST_BINS:${_qt_prefix}/${QT6_INSTALL_BINS}
QT_INSTALL_LIBS:${_qt_prefix}/${QT6_INSTALL_LIBS}
QT_HOST_LIBS:${_qt_prefix}/${QT6_INSTALL_LIBS}
QT_INSTALL_LIBEXECS:${_qmlscanner_dir}
QT_HOST_LIBEXECS:${_qmlscanner_dir}
QT_INSTALL_PLUGINS:${_qt_prefix}/${QT6_INSTALL_PLUGINS}
QT_INSTALL_IMPORTS:${_qt_prefix}/${QT6_INSTALL_QML}
QT_INSTALL_QML:${_qt_prefix}/${QT6_INSTALL_QML}
QT_INSTALL_TRANSLATIONS:${_qt_prefix}/${QT6_INSTALL_TRANSLATIONS}
EOT
}
case \"\$1\" in
  -query) if [ -n \"\$2\" ]; then vars | sed -n \"s/^\$2://p\"; else vars; fi ;;
  -v|-version|--version) echo \"QMake version 3.1\"; echo \"Using Qt version ${Qt6_VERSION}\" ;;
esac
"
  FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)

# Facts the packaging scripts need (paths differ between Homebrew/apt Qt and vcpkg Qt).
set(_qt_deploy_tool "")
if(APPLE AND TARGET Qt6::macdeployqt)
  set(_qt_deploy_tool "$<TARGET_FILE:Qt6::macdeployqt>")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/package-env.sh" CONTENT
"# Generated by CMake (apps/fission3d/CMakeLists.txt); read by packaging/*/package.sh
FISSION_APP_ID='${FISSION_APP_ID}'
FISSION_APP_NAME='${FISSION_APP_NAME}'
FISSION_VERSION='${PROJECT_VERSION}'
FISSION_MACOS_MIN_VERSION='${FISSION_MACOS_MIN_VERSION}'
FISSION_QML_DIR='${CMAKE_CURRENT_SOURCE_DIR}/qml'
FISSION_BINARY_DIR='${CMAKE_BINARY_DIR}'
FISSION_VCPKG_LIB_DIR='${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib'
FISSION_QT_PREFIX='${QT6_INSTALL_PREFIX}'
FISSION_QT_PLUGINS='${QT6_INSTALL_PREFIX}/${QT6_INSTALL_PLUGINS}'
FISSION_MACDEPLOYQT='${_qt_deploy_tool}'
")

if(APPLE)
  set(_info_plist "${CMAKE_CURRENT_BINARY_DIR}/Info.plist")
  configure_file("${FISSION_PACKAGING_DIR}/macos/Info.plist.in" "${_info_plist}" @ONLY)
  set(_icns "${CMAKE_CURRENT_BINARY_DIR}/${FISSION_APP_NAME}.icns")
  fission_make_icns("${_icns}")
  target_sources(fission3d PRIVATE "${_icns}")
  set_source_files_properties("${_icns}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(fission3d PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${_info_plist}"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${FISSION_APP_ID}")
  # vcpkg bundles the platform plugin "cocoa" next to the executable and the loader then looks
  # only there. The offscreen plugin lets tests and CI run the app without a display.
  # The copy has to use the Qt libraries of the bundle, not the Qt installation (relocate.py).
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(TARGET fission3d POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory
            "$<TARGET_BUNDLE_CONTENT_DIR:fission3d>/PlugIns/platforms"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${QT6_INSTALL_PREFIX}/${QT6_INSTALL_PLUGINS}/platforms/libqoffscreen.dylib"
            "$<TARGET_BUNDLE_CONTENT_DIR:fission3d>/PlugIns/platforms/"
    COMMAND "${Python3_EXECUTABLE}" "${FISSION_PACKAGING_DIR}/macos/relocate.py" --plugin
            "$<TARGET_BUNDLE_DIR:fission3d>"
            "$<TARGET_BUNDLE_CONTENT_DIR:fission3d>/PlugIns/platforms/libqoffscreen.dylib"
    VERBATIM)
  install(TARGETS fission3d BUNDLE DESTINATION .)
else()
  set_target_properties(fission3d PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  install(TARGETS fission3d RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

  set(_desktop "${CMAKE_CURRENT_BINARY_DIR}/${FISSION_APP_ID}.desktop")
  set(_metainfo "${CMAKE_CURRENT_BINARY_DIR}/${FISSION_APP_ID}.metainfo.xml")
  configure_file("${FISSION_PACKAGING_DIR}/linux/${FISSION_APP_ID}.desktop.in" "${_desktop}" @ONLY)
  configure_file("${FISSION_PACKAGING_DIR}/linux/${FISSION_APP_ID}.metainfo.xml.in"
                 "${_metainfo}" @ONLY)
  install(FILES "${_desktop}" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
  install(FILES "${_metainfo}" DESTINATION "${CMAKE_INSTALL_DATADIR}/metainfo")
  install(FILES "${FISSION_PACKAGING_DIR}/linux/${FISSION_APP_ID}.xml"
          DESTINATION "${CMAKE_INSTALL_DATADIR}/mime/packages")
  install(FILES "${FISSION_PACKAGING_DIR}/icons/${FISSION_APP_ID}.svg"
          DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
  foreach(_size ${FISSION_ICON_SIZES})
    if(_size LESS_EQUAL 512)
      install(FILES "${FISSION_PACKAGING_DIR}/icons/png/icon-${_size}.png"
              DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/${_size}x${_size}/apps"
              RENAME "${FISSION_APP_ID}.png")
    endif()
  endforeach()
endif()
